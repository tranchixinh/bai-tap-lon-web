  function importManualJson() {
  try {
    const raw = document.getElementById("manualJson").value.trim();
    const post = JSON.parse(raw);

    if (!post.title || !post.url || !post.category) {
      alert("Bài viết cần có title, url và category.");
      return;
    }

    let posts = JSON.parse(localStorage.getItem("posts_data") || "[]");

    const index = posts.findIndex(p => p.id === post.id);

    if (index >= 0) {
      posts[index] = post;
    } else {
      posts.unshift(post);
    }

    localStorage.setItem("posts_data", JSON.stringify(posts));
    alert("Đã lưu bài viết thành công!");
  } catch (error) {
    alert("JSON không hợp lệ. Bạn hãy kiểm tra lại dữ liệu.");
  }
}
